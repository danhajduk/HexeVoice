cmake_minimum_required(VERSION 3.16)

if(DEFINED ENV{HEXE_FIRMWARE_APP} AND NOT "$ENV{HEXE_FIRMWARE_APP}" STREQUAL "")
  set(HEXE_FIRMWARE_APP "$ENV{HEXE_FIRMWARE_APP}" CACHE STRING "Hexe firmware app" FORCE)
elseif(NOT DEFINED HEXE_FIRMWARE_APP OR HEXE_FIRMWARE_APP STREQUAL "")
  set(HEXE_FIRMWARE_APP "endpoint")
endif()

set(HEXE_FIRMWARE_APP_DIR "${CMAKE_CURRENT_LIST_DIR}/apps/${HEXE_FIRMWARE_APP}/main")
if(NOT EXISTS "${HEXE_FIRMWARE_APP_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Unsupported HEXE_FIRMWARE_APP=${HEXE_FIRMWARE_APP}: missing ${HEXE_FIRMWARE_APP_DIR}/CMakeLists.txt")
endif()

if(HEXE_FIRMWARE_APP STREQUAL "endpoint")
  set(HEXE_FIRMWARE_RUNTIME_COMPONENT endpoint_runtime)
elseif(HEXE_FIRMWARE_APP STREQUAL "audio_probe")
  set(HEXE_FIRMWARE_RUNTIME_COMPONENT audio_probe_runtime)
elseif(HEXE_FIRMWARE_APP STREQUAL "recovery")
  set(HEXE_FIRMWARE_RUNTIME_COMPONENT recovery_runtime)
  set(HEXE_FIRMWARE_ADDITIONAL_COMPONENTS espressif__cjson)
  set(HEXE_FIRMWARE_ADDITIONAL_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/managed_components/espressif__cjson")
  if(HEXE_BOARD_PROFILE STREQUAL "waveshare_s3_touch_lcd_1_85c_box_v2")
    set(HEXE_FIRMWARE_DISPLAY_COMPONENT_DIRS
      "${CMAKE_CURRENT_LIST_DIR}/managed_components/espressif__cmake_utilities"
      "${CMAKE_CURRENT_LIST_DIR}/managed_components/espressif__esp_lcd_st77916"
    )
    list(
      APPEND
      HEXE_FIRMWARE_ADDITIONAL_COMPONENTS
        espressif__cmake_utilities
        espressif__esp_lcd_st77916
    )
    foreach(HEXE_FIRMWARE_DISPLAY_COMPONENT_DIR IN LISTS HEXE_FIRMWARE_DISPLAY_COMPONENT_DIRS)
      if(EXISTS "${HEXE_FIRMWARE_DISPLAY_COMPONENT_DIR}/CMakeLists.txt")
        list(APPEND HEXE_FIRMWARE_ADDITIONAL_COMPONENT_DIRS "${HEXE_FIRMWARE_DISPLAY_COMPONENT_DIR}")
      endif()
    endforeach()
  endif()
else()
  message(FATAL_ERROR "Unsupported HEXE_FIRMWARE_APP=${HEXE_FIRMWARE_APP}")
endif()

set(HEXE_FIRMWARE_RUNTIME_DIR "${CMAKE_CURRENT_LIST_DIR}/components/${HEXE_FIRMWARE_RUNTIME_COMPONENT}")
if(NOT EXISTS "${HEXE_FIRMWARE_RUNTIME_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Firmware runtime component ${HEXE_FIRMWARE_RUNTIME_COMPONENT} is missing ${HEXE_FIRMWARE_RUNTIME_DIR}/CMakeLists.txt")
endif()

set(COMPONENT_DIRS
  "$ENV{IDF_PATH}/components"
  "${HEXE_FIRMWARE_APP_DIR}"
  "${HEXE_FIRMWARE_RUNTIME_DIR}"
  ${HEXE_FIRMWARE_ADDITIONAL_COMPONENT_DIRS}
)
set(COMPONENTS main ${HEXE_FIRMWARE_RUNTIME_COMPONENT} ${HEXE_FIRMWARE_ADDITIONAL_COMPONENTS})

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
set(HEXE_DEPENDENCY_LOCK_IDF_VERSION
  "${IDF_VERSION_MAJOR}.${IDF_VERSION_MINOR}.${IDF_VERSION_PATCH}"
)
if(DEFINED HEXE_REQUIRED_IDF_VERSION AND NOT "${HEXE_REQUIRED_IDF_VERSION}" STREQUAL "")
  set(HEXE_DEPENDENCY_LOCK_IDF_VERSION "${HEXE_REQUIRED_IDF_VERSION}")
endif()
set(HEXE_DEPENDENCY_LOCK
  "${CMAKE_CURRENT_LIST_DIR}/dependency-locks/${HEXE_BOARD_PROFILE}-${HEXE_FIRMWARE_APP}-idf-${HEXE_DEPENDENCY_LOCK_IDF_VERSION}.lock"
)
idf_build_set_property(DEPENDENCIES_LOCK "${HEXE_DEPENDENCY_LOCK}")
project(hexe_firmware)

if(HEXE_BOARD_PROFILE STREQUAL "waveshare_p4_wifi6_touch_lcd_7b" AND HEXE_FIRMWARE_APP STREQUAL "endpoint")
  execute_process(
    COMMAND
      "${PYTHON}"
      "${CMAKE_CURRENT_LIST_DIR}/tools/patch-esp-hosted-oom.py"
      "${CMAKE_CURRENT_LIST_DIR}/managed_components/espressif__esp_hosted"
    RESULT_VARIABLE HEXE_ESP_HOSTED_PATCH_RESULT
    ERROR_VARIABLE HEXE_ESP_HOSTED_PATCH_ERROR
  )
  if(NOT HEXE_ESP_HOSTED_PATCH_RESULT EQUAL 0)
    message(FATAL_ERROR "Failed to patch ESP-Hosted DMA OOM handling: ${HEXE_ESP_HOSTED_PATCH_ERROR}")
  endif()
endif()
